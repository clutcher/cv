\documentclass[10pt]{article}
% Define the tightlist command to avoid Pandoc error
\providecommand{\tightlist}{%
    \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Additional packages
\usepackage{lmodern} % Latin Modern fonts
\usepackage{hyperref} % Hyperlinks
\usepackage{titlesec} % For section title formatting
\usepackage{tikz} % For drawing lines
% Source Sans Pro for pdflatex; pdflatex is used because it writes real spaces into the PDF, which LuaLaTeX does not, and some ATS parsers read the text without them
\usepackage[T1]{fontenc}
\usepackage[default]{sourcesanspro}
\input{glyphtounicode}\pdfgentounicode=1
\pdfinterwordspaceon\pdffakespace
\usepackage{enumitem} % For custom lists
\usepackage{color} % For custom colors

% Adjust the page margins
\usepackage[a4paper, margin=1.5cm]{geometry}

% No hyphenation: a split word reads worse and can break keyword matching of ATS parsers
\hyphenpenalty=10000
\exhyphenpenalty=10000
\setlength{\emergencystretch}{3em}

% Adjust left margins and indentations on all elements
\setlength{\parindent}{0pt}
% A markdown blockquote is a block set apart from the text above it, without indent (used for the core expertise line)
\renewenvironment{quote}{\par\medskip}{\par}
\titlespacing*{\subsubsection}{0pt}{3.25ex plus 1ex minus .2ex}{1.5ex plus .2ex}
\setlist[itemize]{
    align=left,
    leftmargin=0pt,
    labelwidth=1.em,
    itemindent=1.em,
    labelsep=0pt,
}

% Define Colors
\definecolor{linecolor}{RGB}{221, 221, 221} % Header line color
\definecolor{accentcolor}{RGB}{56, 115, 179} % Additional accent color

\definecolor{mainfontcolor}{RGB}{115, 115, 115} % Font main color
\definecolor{additionalfontcolor}{RGB}{52, 73, 94} % Font additional color


\color{mainfontcolor}

% Disable numbering on \section, \subsection, \subsubsection (markdown h1, h2, h3)
\makeatletter
\renewcommand{\@seccntformat}[1]{}
\renewcommand{\thesubsection}{}
\renewcommand{\thesubsubsection}{}
\makeatother

% Disable page numbering
\pagenumbering{gobble}

% Change color for bullet list markers
\renewcommand{\labelitemi}{\textcolor{accentcolor}{\textbullet}}

% Display all \emph (markdown italic font) on the right side of same line (used for dates and places)
\let\oldemph\emph
\renewcommand{\emph}[1]{\hfill{\small\normalfont\oldemph{#1}}}

% Redefine \textbf (markdown bold font) with color (used for job title)
\let\oldtextbf\textbf
\renewcommand{\textbf}[1]{\textcolor{additionalfontcolor}{\oldtextbf{#1}}}

% Underline Hyperlinks
% The underline is drawn in the text, not as a PDF link border: a border is often not printed, and its distance from the text changes with descenders
\hypersetup{
    pdfborder={0 0 0},
    pdftitle={Igor Zarvanskyi - Software Architect and Engineering Leader},
    pdfauthor={Igor Zarvanskyi},
    pdfkeywords={Software Architect, Engineering Leader, Engineering Manager, Agentic AI Engineering, Multi-agent Systems, Claude Code, Microservices, Java, Spring Boot, SAP Commerce},
}
\let\oldhref\href
\renewcommand{\href}[2]{\oldhref{#1}{\underline{#2\vphantom{y}}}}

% Professional title under the name, right aligned like the name and contacts (markdown h4); without an h4 nothing is shown
\titleformat{\paragraph}[block]{\raggedleft\Large\color{additionalfontcolor}}{}{0pt}{}
\titlespacing*{\paragraph}{0pt}{0pt}{1.5ex}

% Display \section on right side and adjust font (markdown h1; used for first and last name)
\makeatletter
\renewcommand\section{\@startsection {section}{1}{\z@}%
{-3.5ex \@plus -1ex \@minus -.2ex}%
{2.3ex \@plus.2ex}%
{\normalfont\Huge\bfseries\raggedleft\color{accentcolor}}}
\makeatother

% Add green and grey lines under \subsection and font size (markdown h2; used for main sections work experience etc.)
\titleformat{\subsection}[hang]{\LARGE\color{mainfontcolor}}{}{0pt}{%
    \noindent%
    \tikz[overlay,remember picture,baseline=(a.base)]{%
        \coordinate (a) at (0,0);%
        \draw[line width=1pt, linecolor] (0,-3pt) -- (\linewidth,-3pt);%
        \draw[line width=3pt, accentcolor] (0,-3pt) -- (0.08\linewidth,-3pt);%
    }%
}

% Adjust \subsubsection font size (markdown h3; used for company names)
\titleformat{\subsubsection}[hang]{\Large\bfseries\color{accentcolor}}{}{0pt}{}


% Write URLs in the text font without break points, so no fake spaces appear inside them
\urlstyle{same}
\makeatletter\def\UrlBreaks{}\def\UrlBigBreaks{}\def\UrlNoBreaks{}\def\UrlSpecials{}\makeatother
